pub fn generate_synthetic_data() {
    let mut entries = Vec::new();
    for i in 0..100 {
        let file_path = format!("synthetic_file_{}.rs", i);
        let mod_path = format!("synthetic_mod_{}", i);
        let attributes = vec![parse_quote! { #[synthetic({})] }, i];
        entries.push((file_path, mod_path, attributes));
    }
    println!("Generated {} synthetic entries.", entries.len());
}
